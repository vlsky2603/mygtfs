💻 Примеры кода

// Новая логика позиционирования if (timeToArrivalMinutes > SIMULATION_CONFIG.TIME_THRESHOLDS.VERY_FAR) { // 60+ минут - начало маршрута (0-15%) const startProgress = Math.min(SIMULATION_CONFIG.ROUTE_PROGRESS.VERY_FAR_MAX, uniqueOffset / 200); estimatedPosition = tripStartIndex + Math.floor(routeLength * startProgress); } else if (timeToArrivalMinutes > SIMULATION_CONFIG.TIME_THRESHOLDS.FAR) { // 30-60 минут - первая треть (5-30%) const earlyProgress = SIMULATION_CONFIG.ROUTE_PROGRESS.FAR_MIN + (SIMULATION_CONFIG.ROUTE_PROGRESS.FAR_MAX - SIMULATION_CONFIG.ROUTE_PROGRESS.FAR_MIN) * ((SIMULATION_CONFIG.TIME_THRESHOLDS.VERY_FAR - timeToArrivalMinutes) / SIMULATION_CONFIG.TIME_THRESHOLDS.VERY_FAR); estimatedPosition = tripStartIndex + Math.floor(routeLength * earlyProgress); } // Уникальный сдвиг для избежания наложения const busIdHash = busId.split('').reduce((a, b) => { a = ((a << 5) - a) + b.charCodeAt(0); return a & a; }, 0); const uniqueOffset = Math.abs(busIdHash) % SIMULATION_CONFIG.UNIQUE_OFFSET_RANGE;
